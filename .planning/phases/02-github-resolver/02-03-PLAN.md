       ```rust
       let (token, auth_method) = github::discover_token();
       if matches!(auth_method, github::AuthMethod::Unauthenticated) {
           eprintln!("warning: running unauthenticated (60 requests/hour limit)");
           eprintln!("  hint: set GITHUB_TOKEN or install gh CLI for 5,000 requests/hour");

       if cli.verbose {
       ```

       Note: The `map_err` is needed because reqwest::Error from Client::builder().build() doesn't implement miette::Diagnostic.

    3. Add per-repo processing loop:
       ```rust
       let mut all_actions: Vec<resolver::SyncAction> = Vec::new();
       let mut failed_repos: Vec<(String, String)> = Vec::new(); // (repo, reason)

       for repo_entry in &manifest.repos {
           // Split owner/repo
           let parts: Vec<&str> = repo_entry.splitn(2, '/').collect();
           let (owner, repo_name) = (parts[0], parts[1]);

           // Get repo info (default branch)
           let repo_info = match client.get_repo_info(owner, repo_name) {
               Ok(info) => info,
               Err(e) => {
                   // Check if rate limited — fail immediately
                   if let Some(rl) = e.downcast_ref::<SkillsSyncError>() {
                       if matches!(rl, SkillsSyncError::RateLimited { .. }) {
                           return Err(e);
                       }
                   }
                   eprintln!("warning: {} — {}, skipping", repo_entry, e);
                   failed_repos.push((repo_entry.clone(), e.to_string()));
                   continue;
                }
           };

           if cli.verbose {
               eprintln!("{}: default branch = {}", repo_entry, repo_info.default_branch);
           }

           // Get tree
           let tree = match client.get_tree(owner, repo_name, &repo_info.default_branch) {
               Ok(t) => t,
               Err(e) => {
                   if let Some(rl) = e.downcast_ref::<SkillsSyncError>() {
                       if matches!(rl, SkillsSyncError::RateLimited { .. }) {
                           return Err(e);
                       }
                   }
                   eprintln!("warning: {} — {}, skipping", repo_entry, e);
                   failed_repos.push((repo_entry.clone(), e.to_string()));
                   continue;
               }
           };

           // Detect skills
           let remote_skills = resolver::detect_skills(&tree);
           if remote_skills.is_empty() {
               eprintln!("warning: {} — no SKILL.md found, skipping", repo_entry);
               continue;
           }

           if cli.verbose {
               let skill_type = if remote_skills.len() == 1 && remote_skills[0].subdirectory.is_none() {
                   "single-skill".to_string()
               } else {
                   format!("monorepo ({} skills)", remote_skills.len())
               };
               eprintln!("{}: detected as {}", repo_entry, skill_type);
           }

           // Load local metadata for this repo's skills
           // For now, collect from target directory (default: .claude/skills/)
           let local_metadata = load_local_metadata(repo_entry);

           // Get commit SHA from refs API for accurate tracking
           // Use tree.sha as the reference SHA for now (will be replaced with commit SHA in Phase 3 when we have the target directory)
           let repo_skills = resolver::RepoSkills {
               repo: repo_entry.clone(),
               skills: remote_skills,
               commit_sha: tree.sha.clone(),
           };

           // Resolve
           let actions = resolver::resolve(&repo_skills, &local_metadata);
           all_actions.extend(actions);
       }

       // Summary
       if !failed_repos.is_empty() {
           eprintln!("\nfailed repos:");
           for (repo, reason) in &failed_repos {
               eprintln!("  {} — {}", repo, reason);
           }
       }

       // For now, just report the plan (Phase 3 will execute it)
       let installs = all_actions.iter().filter(|a| matches!(a, resolver::SyncAction::Install { .. })).count();
       let updates = all_actions.iter().filter(|a| matches!(a, resolver::SyncAction::Update { .. })).count();
       let removes = all_actions.iter().filter(|a| matches!(a, resolver::SyncAction::Remove { .. })).count();
       let skips = all_actions.iter().filter(|a| matches!(a, resolver::SyncAction::Skip { .. })).count();

       if installs > 0 || updates > 0 || removes > 0 {
           println!("sync plan: {} install, {} update, {} remove, {} up-to-date", installs, updates, removes, skips);
       } else {
           println!("everything up to date ({} skills)", skips);
       }

       if !failed_repos.is_empty() {
           return Err(SkillsSyncError::GitHubApi {
               status: 0,
               message: format!("{} repo(s) failed", failed_repos.len()),
               repo: failed_repos.iter().map(|(r, _)| r.as_str()).collect::<Vec<_>>().join(", "),
           }.into());
       }

       Ok(())
       ```

    4. Add helper function `load_local_metadata` in main.rs:

       ```rust
       fn load_local_metadata(repo: &str) -> std::collections::HashMap<String, skills_sync::metadata::SkillMeta> {
           // Scan .claude/skills/ for existing metadata files
           // Each skill directory contains .skills-sync-meta.json
           let mut metadata = std::collections::HashMap::new();
           let skills_dir = std::path::Path::new(".claude/skills");
           if !skills_dir.exists() {
               return metadata;
           }
           if let Ok(entries) = std::fs::read_dir(skills_dir) {
               for entry in entries.flatten() {
                   let meta_path = entry.path().join(skills_sync::metadata::META_FILENAME);
                   if meta_path.exists() {
                       if let Ok(meta) = skills_sync::metadata::read_metadata(&meta_path) {
                           if meta.source == repo {
                               let key = skills_sync::resolver::skill_key(repo, &meta.subdirectory);
                               metadata.insert(key, meta);
                           }
                       }
                   }
               }
           }
           metadata
       }
       ```
