5. Add dry-run short-circuit AFTER orphan scan, before `execute_sync_plan`. Per CONTEXT.md:
   - Print each action with prefixes: `+` install, `~` update, `-` remove. Skip actions are omitted.
   - Verbose mode adds SHA details to update lines: `~ owner/repo (update: abcd1234..efgh5678)`
   - Exit code 2 when changes pending, 0 when everything up-to-date.
   - Use `std::process::exit(2)` (safe at this point — no resources to clean up).

```rust
if cli.dry_run {
    for action in &all_actions {
        match action {
            resolver::SyncAction::Install { repo, subdirectory, .. } => {
                let name = resolver::skill_key(repo, subdirectory);
                println!("+ {} (install)", name);
            }
            resolver::SyncAction::Update { repo, subdirectory, old_sha, new_sha, .. } => {
                let name = resolver::skill_key(repo, subdirectory);
                if cli.verbose {
                    println!("~ {} (update: {}..{})", name, &old_sha[..8.min(old_sha.len())], &new_sha[..8.min(new_sha.len())]);
                } else {
                    println!("~ {} (update)", name);
                }
            }
            resolver::SyncAction::Remove { repo, subdirectory, .. } => {
                let name = resolver::skill_key(repo, subdirectory);
                println!("- {} (remove)", name);
            }
            resolver::SyncAction::Skip { .. } => {}
        }
    }
    let pending = all_actions.iter().filter(|a| !matches!(a, resolver::SyncAction::Skip { .. })).count();
    if pending > 0 {
        std::process::exit(2);
    }
    return Ok(());
}
```
