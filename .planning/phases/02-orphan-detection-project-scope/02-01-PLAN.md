2. Create src/sources.ts:
   - Import { readFile, writeFile, rename, readdir } from 'node:fs/promises'
   - Import { join } from 'node:path'
   - Import type { SourcesFile } from './types.js'
   - Constants: SOURCES_FILENAME = '.skills-sync-sources.json', SOURCES_VERSION = 1
   - Export SOURCES_FILENAME for use by other modules

   readSourcesFile(basePath: string = process.cwd()):
   - Build path: join(basePath, SOURCES_FILENAME)
   - Try readFile + JSON.parse
   - If file missing (catch ENOENT): return null
   - If JSON corrupt (catch SyntaxError): return null
   - If parsed.version !== SOURCES_VERSION: return null
   - If valid: return parsed as SourcesFile

   writeSourcesFile(bySource: Map<string, string[]>, basePath: string = process.cwd()):
   - Build the SourcesFile object: { version: SOURCES_VERSION, sources: {} }
   - Iterate bySource entries, normalize keys with normalizeSource(), add to sources record
   - NOTE: Import normalizeSource from './state.js' — the one canonical normalizer
   - Write to tmpPath = filePath + '.tmp' first
   - rename(tmpPath, filePath) for atomic write
   - Use JSON.stringify(data, null, 2) for human-readable output

   detectInstalledSkills(basePath: string = process.cwd()):
   - Scan dirs: ['.agents/skills', '.claude/skills']
   - For each: try readdir with { withFileTypes: true }, filter isDirectory(), collect names
   - Return deduplicated array of skill directory names (use Set)
   - On missing directory: silently continue (try/catch)

     - Ignores files (only directories count)
