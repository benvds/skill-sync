   use crate::error::SkillsSyncError;
   use std::path::{Path, PathBuf};

   const LOCAL_MANIFEST: &str = ".skills.toml";
   const GLOBAL_MANIFEST_DIR: &str = "skills-sync";
   const GLOBAL_MANIFEST_FILE: &str = "skills.toml";

   /// Discover the manifest file from multiple locations.
   /// Priority: --config override > CWD .skills.toml > global ~/.config/skills-sync/skills.toml
   pub fn discover_manifest(config_override: Option<&Path>) -> crate::error::Result<PathBuf> {
       let mut searched = Vec::new();

       // 1. Explicit --config takes highest priority
       if let Some(path) = config_override {
           if path.exists() {
               return Ok(path.to_owned());
           }
           searched.push(path.to_owned());
           return Err(SkillsSyncError::ManifestNotFound { searched }.into());
       }

       // 2. CWD local manifest
       let local = PathBuf::from(LOCAL_MANIFEST);
       if local.exists() {
           return Ok(local);
       }
       searched.push(local);

       // 3. Global XDG config
       if let Some(config_dir) = dirs::config_dir() {
           let global = config_dir.join(GLOBAL_MANIFEST_DIR).join(GLOBAL_MANIFEST_FILE);
           if global.exists() {
               return Ok(global);
           }
           searched.push(global);
       }

       Err(SkillsSyncError::ManifestNotFound { searched }.into())
   }
   ```

   use crate::error::SkillsSyncError;
   use std::path::Path;

   /// The documented sample .skills.toml template.
   /// Tutorial-style with full header, usage section, format explanation,
   /// and commented examples — self-teaching for new users.
   const INIT_TEMPLATE: &str = r#"# ============================================================================
# skills-sync manifest
# ============================================================================
#
# This file tells skills-sync which skill repos to install.
# Run `skills-sync` to sync all listed repos to your skills directory.
#
# ── Usage ────────────────────────────────────────────────────────────────────
#
#   skills-sync              Sync all repos listed below
#   skills-sync --dry-run    Preview changes without applying
#   skills-sync --verbose    Show detailed sync output
#   skills-sync --global     Sync to ~/.claude/skills/ instead of local
#
# ── Format ───────────────────────────────────────────────────────────────────
#
#   Repos use "owner/repo" format (GitHub repos):
#
#     [skills]
#     repos = [
#         "owner/single-skill-repo",    # repo with SKILL.md at root
#         "owner/multi-skill-monorepo", # repo with skill subdirectories
#     ]
#
#   Single-skill repos have a SKILL.md at the repository root.
#   Monorepos contain subdirectories, each with their own SKILL.md.
#   skills-sync detects the layout automatically.
#
# ── Examples ─────────────────────────────────────────────────────────────────
#
#   Uncomment the repos below to try them, or add your own:
#
#     # "anthropics/claude-skills"      # single-skill repo example
#     # "example-org/skill-collection"  # monorepo with multiple skills
#
# ============================================================================

[skills]
repos = [
    # Add your skill repos here, one per line:
    # "owner/repo",
]
"#;

   /// Generate a sample .skills.toml in the given directory.
   pub fn run_init(path: &Path) -> crate::error::Result<()> {
       if path.exists() {
           return Err(SkillsSyncError::ManifestAlreadyExists {
               path: path.to_owned(),
           }.into());
       }

       std::fs::write(path, INIT_TEMPLATE)
           .map_err(SkillsSyncError::Io)?;

       eprintln!("Created {}", path.display());
       Ok(())
   }
   ```

3. Update src/main.rs to wire everything together:
   ```rust
   use clap::Parser;
   use skills_sync::cli::Cli;
   use skills_sync::config::discover_manifest;
   use skills_sync::init::run_init;
   use skills_sync::manifest::parse_and_validate;

   fn main() {
       // Install miette error handler for fancy diagnostics
       let _ = miette::set_hook(Box::new(|_| {
           Box::new(miette::MietteHandlerOpts::new()
               .terminal_links(true)
               .unicode(true)
               .context_lines(2)
               .build())
       }));

       let cli = Cli::parse();

       if let Err(err) = run(cli) {
           eprintln!("{:?}", err);
           std::process::exit(1);
       }
   }

   fn run(cli: Cli) -> skills_sync::error::Result<()> {
       if cli.init {
           let path = std::path::PathBuf::from(".skills.toml");
           return run_init(&path);
       }

       let manifest_path = discover_manifest(cli.config.as_deref())?;
       let content = std::fs::read_to_string(&manifest_path)
           .map_err(skills_sync::error::SkillsSyncError::Io)?;
       let _manifest = parse_and_validate(&content, &manifest_path)?;

       Ok(())
   }
   ```
